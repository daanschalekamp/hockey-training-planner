<script>

(function(){

"use strict";


const FIELD_W = 500;

const FIELD_H_FULL = 800;

const FIELD_H_HALF = 400;

const COLORS = { blue:'#3f7de0', red:'#e0453f', yellow:'#e0c23f' };

const STORE_KEY_EX = 'htp_exercises_v1';

const STORE_KEY_CUR = 'htp_current_v1';


let exercises = [];

let currentId = null;

let tool = 'select';

let playerColor = 'blue';

let drawColor = '#f3f7f1';

let drawDashed = false;

let arrowDashed = false;

let arrowColor = '#f3f7f1';

let selectedId = null;


let scaleZoom = 1;

let baseScale = 1;

let panX = 0, panY = 0;

let dpr = Math.max(1, window.devicePixelRatio || 1);


let activePointers = new Map();

let pinch = null;

let panState = null;

let dragItem = null;

let resizeState = null;

let rotateState = null;

let arrowHandleDrag = null;

let pendingArrow = null;

let pendingStroke = null;

let lastTap = {id:null, t:0};

let history = [];


const canvas = document.getElementById('field');

const ctx = canvas.getContext('2d');


function uid(){ return Math.random().toString(36).slice(2,10) + Date.now().toString(36); }

function defaultExercise(name){

  return { id: uid(), name: name || 'New Exercise', mode:'full', items: [], strokes: [] };

}

function loadAll(){

  try{

    const raw = localStorage.getItem(STORE_KEY_EX);

    exercises = raw ? JSON.parse(raw) : [];

  }catch(e){ exercises = []; }

  if(!exercises.length){ exercises.push(defaultExercise('Warm-up 1')); }

  currentId = localStorage.getItem(STORE_KEY_CUR) || exercises[0].id;

  if(!exercises.find(e=>e.id===currentId)) currentId = exercises[0].id;

}

function saveAll(){

  localStorage.setItem(STORE_KEY_EX, JSON.stringify(exercises));

  localStorage.setItem(STORE_KEY_CUR, currentId);

}

function current(){ return exercises.find(e=>e.id===currentId); }


function snapshot(){

  const c = current();

  history.push(JSON.stringify({items:c.items, strokes:c.strokes}));

  if(history.length > 25) history.shift();

}

function undo(){

  if(!history.length){ toast('Nothing to undo'); return; }

  const data = JSON.parse(history.pop());

  const c = current();

  c.items = data.items; c.strokes = data.strokes;

  selectedId = null;

  saveAll(); render(); renderContextBar();

}


let toastTimer=null;

function toast(msg){

  const el = document.getElementById('toast');

  el.textContent = msg;

  el.classList.add('show');

  clearTimeout(toastTimer);

  toastTimer = setTimeout(()=>el.classList.remove('show'), 1600);

}


function fieldHeight(mode){ return mode==='full' ? FIELD_H_FULL : FIELD_H_HALF; }

function circlePt(cx,cy,r,deg){

  const a = deg*Math.PI/180;

  return {x:cx+r*Math.cos(a), y:cy+r*Math.sin(a)};

}


function drawD(ctx, centerX, goalY, R, halfW, bulge){

  const lp = {x:centerX-halfW, y:goalY};

  const rp = {x:centerX+halfW, y:goalY};

  ctx.beginPath();

  if(bulge === -1){

    ctx.moveTo(lp.x - R, lp.y);

    ctx.arc(lp.x, lp.y, R, Math.PI, 1.5*Math.PI, false);

    ctx.lineTo(rp.x, rp.y - R);

    ctx.arc(rp.x, rp.y, R, 1.5*Math.PI, 2*Math.PI, false);

    ctx.lineTo(rp.x, rp.y);

    ctx.lineTo(lp.x, lp.y);

  } else {

    ctx.moveTo(lp.x - R, lp.y);

    ctx.arc(lp.x, lp.y, R, Math.PI, 0.5*Math.PI, true);

    ctx.lineTo(rp.x, rp.y + R);

    ctx.arc(rp.x, rp.y, R, 0.5*Math.PI, 0, true);

    ctx.lineTo(rp.x, rp.y);

    ctx.lineTo(lp.x, lp.y);

  }

  ctx.closePath();

  ctx.stroke();

}


function drawGoalCage(ctx, centerX, goalY, halfW, dir){

  const depth = 14;

  ctx.strokeStyle = '#dfe9e2';

  ctx.lineWidth = 2.4;

  ctx.beginPath();

  ctx.moveTo(centerX-halfW, goalY);

  ctx.lineTo(centerX-halfW, goalY - dir*depth);

  ctx.lineTo(centerX+halfW, goalY - dir*depth);

  ctx.lineTo(centerX+halfW, goalY);

  ctx.stroke();

  ctx.lineWidth = 0.8;

  ctx.strokeStyle = 'rgba(223,233,226,0.45)';

  const steps = 6;

  for(let i=1;i<steps;i++){

    const gx = centerX - halfW + (2*halfW)*(i/steps);

    ctx.beginPath();

    ctx.moveTo(gx, goalY);

    ctx.lineTo(gx, goalY - dir*depth);

    ctx.stroke();

  }

}


function drawField(ctx, mode){

  const fh = fieldHeight(mode);

  const margin = 12;

  ctx.fillStyle = '#0f4326';

  ctx.fillRect(0,0,FIELD_W,fh);

  const bands = mode==='full' ? 16 : 8;

  const bandH = fh/bands;

  for(let i=0;i<bands;i++){

    ctx.fillStyle = i%2===0 ? 'rgba(255,255,255,0.025)' : 'rgba(0,0,0,0.05)';

    ctx.fillRect(0, i*bandH, FIELD_W, bandH);

  }

  ctx.lineWidth = 2.2;

  ctx.strokeStyle = '#f3f7f1';

  ctx.lineCap = 'butt';

  ctx.strokeRect(margin, margin, FIELD_W-2*margin, fh-2*margin);

  const R = 118;

  const halfGoalW = 16;

  const centerX = FIELD_W/2;

  if(mode === 'full'){

    ctx.beginPath();

    ctx.moveTo(margin, fh/2); ctx.lineTo(FIELD_W-margin, fh/2);

    ctx.stroke();

    ctx.beginPath(); ctx.arc(centerX, fh/2, 3.2, 0, Math.PI*2); ctx.fillStyle='#f3f7f1'; ctx.fill();

    const y23a = margin + (fh-2*margin)*0.25;

    const y23b = fh - margin - (fh-2*margin)*0.25;

    [y23a, y23b].forEach(y=>{

      ctx.save();

      ctx.setLineDash([9,7]);

      ctx.beginPath(); ctx.moveTo(margin,y); ctx.lineTo(FIELD_W-margin,y); ctx.stroke();

      ctx.restore();

    });

    drawD(ctx, centerX, fh-margin, R, halfGoalW, -1);

    drawD(ctx, centerX, margin, R, halfGoalW, +1);

    drawGoalCage(ctx, centerX, fh-margin, halfGoalW, -1);

    drawGoalCage(ctx, centerX, margin, halfGoalW, +1);

    ctx.beginPath(); ctx.arc(centerX, fh-margin-58, 2.6,0,Math.PI*2); ctx.fillStyle='#f3f7f1'; ctx.fill();

    ctx.beginPath(); ctx.arc(centerX, margin+58, 2.6,0,Math.PI*2); ctx.fillStyle='#f3f7f1'; ctx.fill();

  } else {

    ctx.save();

    ctx.setLineDash([9,7]);

    ctx.beginPath(); ctx.moveTo(margin, margin); ctx.lineTo(FIELD_W-margin, margin); ctx.stroke();

    ctx.restore();

    const y23 = margin + (fh-2*margin)*0.42;

    ctx.save();

    ctx.setLineDash([9,7]);

    ctx.beginPath(); ctx.moveTo(margin,y23); ctx.lineTo(FIELD_W-margin,y23); ctx.stroke();

    ctx.restore();

    drawD(ctx, centerX, fh-margin, R, halfGoalW, -1);

    drawGoalCage(ctx, centerX, fh-margin, halfGoalW, -1);

    ctx.beginPath(); ctx.arc(centerX, fh-margin-58, 2.6,0,Math.PI*2); ctx.fillStyle='#f3f7f1'; ctx.fill();

  }

}


function itemRadius(item){

  switch(item.type){

    case 'player': return 20*item.size;

    case 'cone': return 15*item.size;

    case 'ball': return 9*item.size;

    case 'goal': return 24*item.size;

    default: return 16*item.size;

  }

}


function drawPlayer(ctx,item){

  const R = itemRadius(item);

  ctx.save();

  ctx.translate(item.x,item.y);

  ctx.beginPath(); ctx.ellipse(0,R*0.85,R*0.85,R*0.28,0,0,Math.PI*2);

  ctx.fillStyle='rgba(0,0,0,0.28)'; ctx.fill();

  ctx.beginPath(); ctx.arc(0,0,R,0,Math.PI*2);

  ctx.fillStyle = COLORS[item.color] || item.color;

  ctx.fill();

  ctx.lineWidth = Math.max(1.6, R*0.09);

  ctx.strokeStyle = '#0b1310';

  ctx.stroke();

  ctx.save();

  ctx.rotate((item.rotation||0)*Math.PI/180);

  ctx.beginPath();

  ctx.moveTo(0,-R*1.28);

  ctx.lineTo(R*0.28,-R*0.86);

  ctx.lineTo(-R*0.28,-R*0.86);

  ctx.closePath();

  ctx.fillStyle = '#f3f7f1';

  ctx.fill();

  ctx.restore();

  if(item.label){

    ctx.fillStyle = '#0b1310';

    ctx.font = `800 ${Math.max(10,R*0.95)}px -apple-system, sans-serif`;

    ctx.textAlign='center'; ctx.textBaseline='middle';

    ctx.fillText(String(item.label).slice(0,3), 0, R*0.06);

  }

  ctx.restore();

}


function drawCone(ctx,item){

  const R = itemRadius(item);

  ctx.save();

  ctx.translate(item.x,item.y);

  ctx.rotate((item.rotation||0)*Math.PI/180);

  ctx.beginPath();

  ctx.moveTo(0,-R);

  ctx.lineTo(R*0.82,R*0.75);

  ctx.lineTo(-R*0.82,R*0.75);

  ctx.closePath();

  ctx.fillStyle = '#ff7a1a';

  ctx.fill();

  ctx.lineWidth = 1.6; ctx.strokeStyle='#8f3d00'; ctx.stroke();

  ctx.beginPath();

  ctx.moveTo(-R*0.46,R*0.2); ctx.lineTo(R*0.46,R*0.2);

  ctx.strokeStyle='#fff'; ctx.lineWidth=Math.max(1.2,R*0.16); ctx.stroke();

  ctx.restore();

}


function drawBall(ctx,item){

  const R = itemRadius(item);

  ctx.save();

  ctx.translate(item.x,item.y);

  ctx.beginPath(); ctx.arc(0,0,R,0,Math.PI*2);

  ctx.fillStyle='#f6f2e6'; ctx.fill();

  ctx.lineWidth=1.2; ctx.strokeStyle='#8a7d55'; ctx.stroke();

  ctx.strokeStyle='rgba(138,125,85,0.7)'; ctx.lineWidth=0.9;

  ctx.beginPath(); ctx.arc(0,0,R*0.55,0.3,2.6); ctx.stroke();

  ctx.beginPath(); ctx.arc(0,0,R*0.55,Math.PI+0.3,Math.PI+2.6); ctx.stroke();

  ctx.restore();

}


function drawGoal(ctx,item){

  const s = item.size;

  const w = 46*s, h = 20*s;

  ctx.save();

  ctx.translate(item.x,item.y);

  ctx.rotate((item.rotation||0)*Math.PI/180);

  ctx.fillStyle='rgba(243,247,241,0.16)';

  ctx.fillRect(-w/2,-h/2,w,h);

  ctx.lineWidth=2.4; ctx.strokeStyle='#f3f7f1';

  ctx.strokeRect(-w/2,-h/2,w,h);

  ctx.lineWidth=0.8; ctx.strokeStyle='rgba(243,247,241,0.55)';

  const cols=6;

  for(let i=1;i<cols;i++){

    const gx = -w/2 + w*(i/cols);

    ctx.beginPath(); ctx.moveTo(gx,-h/2); ctx.lineTo(gx,h/2); ctx.stroke();

  }

  ctx.beginPath(); ctx.moveTo(-w/2,0); ctx.lineTo(w/2,0); ctx.stroke();

  ctx.restore();

}


function arrowGeom(item){

  const mx=(item.x1+item.x2)/2, my=(item.y1+item.y2)/2;

  const dx=item.x2-item.x1, dy=item.y2-item.y1;

  const len=Math.hypot(dx,dy)||1;

  const nx=-dy/len, ny=dx/len;

  const curve = item.curve||0;

  const cpx = mx + nx*curve, cpy = my + ny*curve;

  return {mx,my,cpx,cpy};

}


function drawArrow(ctx,item){

  const {cpx,cpy} = arrowGeom(item);

  ctx.save();

  ctx.strokeStyle = item.color;

  ctx.fillStyle = item.color;

  ctx.lineWidth = 3.4;

  ctx.lineCap='round';

  if(item.dashed) ctx.setLineDash([2,10]); else ctx.setLineDash([]);

  ctx.beginPath();

  ctx.moveTo(item.x1,item.y1);

  ctx.quadraticCurveTo(cpx,cpy,item.x2,item.y2);

  ctx.stroke();

  ctx.setLineDash([]);

  const ang = Math.atan2(item.y2-cpy, item.x2-cpx);

  const ah = 11;

  ctx.beginPath();

  ctx.moveTo(item.x2, item.y2);

  ctx.lineTo(item.x2 - ah*Math.cos(ang-0.4), item.y2 - ah*Math.sin(ang-0.4));

  ctx.lineTo(item.x2 - ah*Math.cos(ang+0.4), item.y2 - ah*Math.sin(ang+0.4));

  ctx.closePath();

  ctx.fill();

  ctx.restore();

}


function drawItem(ctx,item){

  if(item.type==='player') drawPlayer(ctx,item);

  else if(item.type==='cone') drawCone(ctx,item);

  else if(item.type==='ball') drawBall(ctx,item);

  else if(item.type==='goal') drawGoal(ctx,item);

  else if(item.type==='arrow') drawArrow(ctx,item);

}


let handles = null;

function computeHandles(item){

  if(!item) return null;

  if(item.type==='arrow'){

    const g = arrowGeom(item);

    return { type:'arrow', start:{x:item.x1,y:item.y1}, end:{x:item.x2,y:item.y2}, curveH:{x:g.cpx,y:g.cpy} };

  }

  const R = itemRadius(item);

  const resize = circlePt(item.x,item.y, R*1.42, 45);

  const rotate = {x:item.x, y:item.y - R - 26};

  return { type:'basic', resize, rotate };

}


function effScale(){ return baseScale*scaleZoom; }


function drawHandlesOverlay(ctx,item){

  if(!item || !handles) return;

  ctx.save();

  ctx.lineWidth = 1.4/effScale();

  ctx.strokeStyle = 'rgba(255,138,61,0.95)';

  ctx.setLineDash([5/effScale(),4/effScale()]);

  if(handles.type==='basic'){

    const R = itemRadius(item)*1.42;

    ctx.beginPath(); ctx.arc(item.x,item.y,R,0,Math.PI*2); ctx.stroke();

    ctx.setLineDash([]);

    ctx.beginPath(); ctx.moveTo(item.x,item.y); ctx.lineTo(handles.rotate.x,handles.rotate.y); ctx.stroke();

    drawHandleDot(ctx,handles.rotate.x,handles.rotate.y,'#ff8a3d','↻');

    drawHandleDot(ctx,handles.resize.x,handles.resize.y,'#4d8dff','⤢');

  } else {

    ctx.setLineDash([]);

    drawHandleDot(ctx,handles.start.x,handles.start.y,'#ff8a3d','');

    drawHandleDot(ctx,handles.end.x,handles.end.y,'#ff8a3d','');

    drawHandleDot(ctx,handles.curveH.x,handles.curveH.y,'#4d8dff','');

  }

  ctx.restore();

}


function drawHandleDot(ctx,x,y,color,glyph){

  const r = 11/effScale();

  ctx.save();

  ctx.beginPath(); ctx.arc(x,y,r,0,Math.PI*2);

  ctx.fillStyle = color; ctx.fill();

  ctx.lineWidth = 1.5/effScale(); ctx.strokeStyle='#0b1310'; ctx.stroke();

  if(glyph){

    ctx.fillStyle='#0b1310';

    ctx.font = `${11/effScale()}px sans-serif`;

    ctx.textAlign='center'; ctx.textBaseline='middle';

    ctx.fillText(glyph,x,y+0.5/effScale());

  }

  ctx.restore();

}


function drawScene(ctx, mode, items, strokes){

  drawField(ctx, mode);

  strokes.forEach(s=>{

    if(s.points.length<2) return;

    ctx.save();

    ctx.strokeStyle=s.color; ctx.lineWidth=3.2;

    ctx.lineJoin='round'; ctx.lineCap='round';

    if(s.dashed) ctx.setLineDash([2,9]);

    ctx.beginPath();

    ctx.moveTo(s.points[0].x, s.points[0].y);

    for(let i=1;i<s.points.length;i++) ctx.lineTo(s.points[i].x, s.points[i].y);

    ctx.stroke();

    ctx.restore();

  });

  items.forEach(it=>drawItem(ctx,it));

}


function fitField(){

  const rect = canvas.getBoundingClientRect();

  canvas.width = Math.round(rect.width*dpr);

  canvas.height = Math.round(rect.height*dpr);

  const c = current();

  const fh = fieldHeight(c.mode);

  baseScale = Math.min(rect.width/(FIELD_W+24), rect.height/(fh+24));

  panX = (rect.width - FIELD_W*effScale())/2;

  panY = (rect.height - fh*effScale())/2;

}


function screenToField(clientX, clientY){

  const rect = canvas.getBoundingClientRect();

  const cx = clientX-rect.left, cy = clientY-rect.top;

  return { x:(cx-panX)/effScale(), y:(cy-panY)/effScale() };

}


function render(){

  const c = current();

  const rect = canvas.getBoundingClientRect();

  ctx.setTransform(dpr,0,0,dpr,0,0);

  ctx.clearRect(0,0,rect.width,rect.height);

  ctx.fillStyle = '#0b1310';

  ctx.fillRect(0,0,rect.width,rect.height);

  ctx.save();

  ctx.translate(panX,panY);

  ctx.scale(effScale(),effScale());


  drawScene(ctx, c.mode, c.items, c.strokes);


  if(pendingArrow){

    ctx.save();

    ctx.strokeStyle = arrowColor; ctx.lineWidth=3;

    if(arrowDashed) ctx.setLineDash([2,10]);

    ctx.beginPath(); ctx.moveTo(pendingArrow.x1,pendingArrow.y1); ctx.lineTo(pendingArrow.x2,pendingArrow.y2); ctx.stroke();

    ctx.restore();

  }

  if(pendingStroke && pendingStroke.points.length>1){

    ctx.save();

    ctx.strokeStyle=pendingStroke.color; ctx.lineWidth=3.2; ctx.lineJoin='round'; ctx.lineCap='round';

    if(pendingStroke.dashed) ctx.setLineDash([2,9]);

    ctx.beginPath();

    ctx.moveTo(pendingStroke.points[0].x,pendingStroke.points[0].y);

    for(let i=1;i<pendingStroke.points.length;i++) ctx.lineTo(pendingStroke.points[i].x,pendingStroke.points[i].y);

    ctx.stroke();

    ctx.restore();

  }


  const sel = c.items.find(i=>i.id===selectedId);

  handles = computeHandles(sel);

  if(sel) drawHandlesOverlay(ctx, sel);


  ctx.restore();

  document.getElementById('zoom-label').textContent = Math.round(scaleZoom*100)+'%';

}


function addItem(type, x, y){

  snapshot();

  const c = current();

  let item = { id:uid(), type, x, y, size:1, rotation:0 };

  if(type==='player'){ item.color = playerColor; item.label=''; }

  if(type==='goal'){ item.color='#fff'; }

  c.items.push(item);

  selectedId = item.id;

  saveAll();

  return item;

}


function hitTestItems(items, x, y){

  for(let i=items.length-1;i>=0;i--){

    const it = items[i];

    let r;

    if(it.type==='arrow'){

      const g = arrowGeom(it);

      let best = Infinity;

      for(let t=0;t<=1;t+=0.05){

        const px = (1-t)*(1-t)*it.x1 + 2*(1-t)*t*g.cpx + t*t*it.x2;

        const py = (1-t)*(1-t)*it.y1 + 2*(1-t)*t*g.cpy + t*t*it.y2;

        best = Math.min(best, Math.hypot(px-x,py-y));

      }

      if(best < 14/effScale()) return it;

      continue;

    }

    r = itemRadius(it)*1.15;

    if(Math.hypot(it.x-x, it.y-y) <= r) return it;

  }

  return null;

}


canvas.addEventListener('pointerdown', (e)=>{

  canvas.setPointerCapture(e.pointerId);

  activePointers.set(e.pointerId, {x:e.clientX, y:e.clientY});


  if(activePointers.size === 2){

    const pts = [...activePointers.values()];

    const rect = canvas.getBoundingClientRect();

    pinch = {

      startDist: Math.hypot(pts[0].x-pts[1].x, pts[0].y-pts[1].y),

      startScale: scaleZoom,

      cx: (pts[0].x+pts[1].x)/2 - rect.left,

      cy: (pts[0].y+pts[1].y)/2 - rect.top,

      startPanX: panX, startPanY: panY

    };

    panState = null; dragItem=null; resizeState=null; rotateState=null; arrowHandleDrag=null;

    pendingArrow=null; pendingStroke=null;

    return;

  }

  if(activePointers.size > 2) return;


  const c = current();

  const fp = screenToField(e.clientX, e.clientY);


  if(tool === 'select'){

    const sel = c.items.find(i=>i.id===selectedId);

    const hr = 16/effScale();

    if(sel && handles){

      if(handles.type==='basic'){

        if(Math.hypot(fp.x-handles.resize.x, fp.y-handles.resize.y) < hr){ resizeState = {id:sel.id, baseDist:Math.hypot(sel.x-handles.resize.x,sel.y-handles.resize.y), baseSize:sel.size}; return; }

        if(Math.hypot(fp.x-handles.rotate.x, fp.y-handles.rotate.y) < hr){ rotateState = {id:sel.id}; return; }

      } else {

        if(Math.hypot(fp.x-handles.start.x, fp.y-handles.start.y) < hr){ arrowHandleDrag={id:sel.id, which:'start'}; return; }

        if(Math.hypot(fp.x-handles.end.x, fp.y-handles.end.y) < hr){ arrowHandleDrag={id:sel.id, which:'end'}; return; }

        if(Math.hypot(fp.x-handles.curveH.x, fp.y-handles.curveH.y) < hr){ arrowHandleDrag={id:sel.id, which:'curve'}; return; }

      }

    }

    const hit = hitTestItems(c.items, fp.x, fp.y);

    if(hit){

      const now = Date.now();

      if(hit.type==='player' && lastTap.id===hit.id && now-lastTap.t < 350){

        const val = prompt('Jersey number / label', hit.label||'');

        if(val !== null){ snapshot(); hit.label = val.slice(0,3); saveAll(); }

        lastTap = {id:null,t:0};

      } else {

        lastTap = {id:hit.id, t:now};

      }

      selectedId = hit.id;

      snapshot();

      dragItem = { id:hit.id, offX: fp.x-hit.x, offY: fp.y-hit.y };

      renderContextBar(); render();

    } else {

      selectedId = null;

      panState = { startX:e.clientX, startY:e.clientY, startPanX:panX, startPanY:panY };

      renderContextBar(); render();

    }

    return;

  }


  if(['player','cone','ball','goal'].includes(tool)){

    addItem(tool, fp.x, fp.y);

    tool = 'select';

    setActiveToolUI();

    renderContextBar();

    render();

    return;

  }


  if(tool === 'arrow'){

    pendingArrow = { x1:fp.x, y1:fp.y, x2:fp.x, y2:fp.y };

    return;

  }


  if(tool === 'draw'){

    pendingStroke = { points:[fp], color:drawColor, dashed:drawDashed };

    return;

  }

});


canvas.addEventListener('pointermove', (e)=>{

  if(!activePointers.has(e.pointerId)) return;

  activePointers.set(e.pointerId, {x:e.clientX, y:e.clientY});


  if(activePointers.size===2 && pinch){

    const pts = [...activePointers.values()];

    const d = Math.hypot(pts[0].x-pts[1].x, pts[0].y-pts[1].y);

    const newScale = Math.min(3, Math.max(0.4, pinch.startScale * (d/pinch.startDist)));

    const factor = newScale/pinch.startScale;

    scaleZoom = newScale;

    panX = pinch.cx - (pinch.cx - pinch.startPanX) * factor;

    panY = pinch.cy - (pinch.cy - pinch.startPanY) * factor;

    render();

    return;

  }


  const c = current();

  const fp = screenToField(e.clientX, e.clientY);


  if(panState){

    panX = panState.startPanX + (e.clientX - panState.startX);

    panY = panState.startPanY + (e.clientY - panState.startY);

    render();

    return;

  }


  if(resizeState){

    const it = c.items.find(i=>i.id===resizeState.id);

    if(it){

      const distNow = Math.hypot(fp.x-it.x, fp.y-it.y);

      it.size = Math.max(0.5, Math.min(2.5, resizeState.baseSize * (distNow/resizeState.baseDist)));

      render();

    }

    return;

  }


  if(rotateState){

    const it = c.items.find(i=>i.id===rotateState.id);

    if(it){

      const rad = Math.atan2(fp.y - it.y, fp.x - it.x);

      it.rotation = Math.round((rad * 180 / Math.PI) + 90);

      render();

    }

    return;

  }


  if(arrowHandleDrag){

    const it = c.items.find(i=>i.id===arrowHandleDrag.id);

    if(it){

      if(arrowHandleDrag.which==='start'){ it.x1=fp.x; it.y1=fp.y; }

      else if(arrowHandleDrag.which==='end'){ it.x2=fp.x; it.y2=fp.y; }

      else if(arrowHandleDrag.which==='curve'){

        const mx=(it.x1+it.x2)/2, my=(it.y1+it.y2)/2;

        const dx=it.x2-it.x1, dy=it.y2-it.y1;

        const len=Math.hypot(dx,dy)||1;

        const nx=-dy/len, ny=dx/len;

        it.curve = (fp.x-mx)*nx + (fp.y-my)*ny;

      }

      render();

    }

    return;

  }


  if(dragItem){

    const it = c.items.find(i=>i.id===dragItem.id);

    if(it){

      it.x = fp.x - dragItem.offX;

      it.y = fp.y - dragItem.offY;

      render();

    }

    return;

  }


  if(pendingArrow){

    pendingArrow.x2 = fp.x;

    pendingArrow.y2 = fp.y;

    render();

    return;

  }


  if(pendingStroke){

    pendingStroke.points.push(fp);

    render();

    return;

  }

});


function endPointer(e){

  activePointers.delete(e.pointerId);

  if(activePointers.size < 2) pinch = null;

  if(activePointers.size === 0){

    if(panState || dragItem || resizeState || rotateState || arrowHandleDrag){

      saveAll();

    }

    if(pendingArrow){

      snapshot();

      const c = current();

      const dx = pendingArrow.x2 - pendingArrow.x1;

      const dy = pendingArrow.y2 - pendingArrow.y1;

      if(Math.hypot(dx,dy) > 12){

        c.items.push({

          id:uid(), type:'arrow',

          x1:pendingArrow.x1, y1:pendingArrow.y1,

          x2:pendingArrow.x2, y2:pendingArrow.y2,

          curve:0, color:arrowColor, dashed:arrowDashed

        });

        saveAll();

      }

      pendingArrow = null;

      render();

    }

    if(pendingStroke){

      snapshot();

      const c = current();

      if(pendingStroke.points.length > 2){

        c.strokes.push(pendingStroke);

        saveAll();

      }

      pendingStroke = null;

      render();

    }

    panState = null; dragItem = null; resizeState = null; rotateState = null; arrowHandleDrag = null;

  }

}


canvas.addEventListener('pointerup', endPointer);

      clone.name = ex.name + ' (Copy)';

      exercises.push(clone);

      currentId = clone.id;

      saveAll(); renderSidebar(); fitField(); updateModeUI(); render();

    };

    btns[1].onclick = (e)=>{

      e.stopPropagation();

      if(exercises.length<=1){ toast('Keep at least one exercise'); return; }

      if(confirm(`Delete "${ex.name}"?`)){

        exercises = exercises.filter(x=>x.id!==ex.id);

        if(currentId===ex.id) currentId = exercises[0].id;

        saveAll(); renderSidebar(); fitField(); updateModeUI(); render();

      }

    };

    list.appendChild(item);

  });

}
